\begin{reporttimeline}[tracks={Event time,Arrival time,Watermark},span=12.0,track gap=1.65]
  % ReportKit's \watermark label styling ("rk transition label") bakes in a
  % TikZ `midway` key intended for edge labels attached to a path. `\watermark`
  % places a standalone node (not a path-attached one), and `midway` on a
  % standalone node silently discards the requested coordinate, collapsing
  % every label onto the picture origin regardless of label-position/track/
  % position. Locally clearing that style to drop `midway` (scoped to this
  % reporttimeline's \begingroup/\endgroup, so it does not leak into other
  % figures) restores the documented label-position behavior without
  % modifying report-kit itself.
  \tikzset{rk transition label/.style={font=\sffamily\fontsize{7.0}{8.2}\selectfont,text=Muted,fill=Surface,inner sep=1.5pt}}
  \event{Event time}{0.12}{02:14 trade}
  \event{Event time}{0.55}{03:02 trade}
  \event{Arrival time}{0.60}{03:03 late}
  \event{Arrival time}{0.88}{03:08 on time}
  \watermark[label-position=below,label-width=26mm]{Watermark}{0.70}{02:59 watermark: the 02:14 trade is still inside allowed lateness}
  \watermark[label-position=below,label-width=24mm]{Watermark}{0.95}{watermark passes 03:00 and closes the 02:00 window}
  \skewarrow{Event time}{0.12}{Arrival time}{0.60}
\end{reporttimeline}
