  \transition[route={bend left=25},label-position=above]{running.north}{retrying.south}{retry}

  \transition[route={bend right=12},label-position=below]{running}{failed}{deterministic error}
  \transition[route={bend right=12},label-position=below]{running}{skipped}{skip}
